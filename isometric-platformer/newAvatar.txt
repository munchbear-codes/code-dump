// 5. Draw Player Character Mesh - UPGRADED TO HUMANOID BATTLEBLOCK ARCADE STYLE WITH TRANSPARENT I-FRAME FLICKERS
    let shouldDrawPlayer = true;
    if (player.invincibilityFrames > 0) {
        if (Math.floor(player.invincibilityFrames / 6) % 2 === 0) {
            shouldDrawPlayer = false;
        }
    }

    if (shouldDrawPlayer) {
        ctx.save();
        // Translate right to the center of the player object box
        ctx.translate(player.x + player.width / 2, player.y + player.height / 2);
        
        // Continuous organic breathing bobbing animation loop tracking
        const playerBob = player.grounded ? Math.sin(Date.now() / 100) * 2 : 0;
        
        // Body Base Outline Profiles
        ctx.fillStyle = player.color;
        ctx.strokeStyle = '#000000';
        ctx.lineWidth = 3;
        
        // Rounded Rectangle Torso
        ctx.beginPath();
        ctx.roundRect(-player.width / 2, -player.height / 2 + 10, player.width, player.height - 10, 10);
        ctx.fill();
        ctx.stroke();

        // Spherical Head Module
        ctx.beginPath();
        ctx.arc(0, -player.height / 2 + 5 + playerBob, 20, 0, Math.PI * 2);
        ctx.fill();
        ctx.stroke();

        // Eyes (BattleBlock Style: Big high-contrast white tracking circles)
        ctx.fillStyle = '#ffffff';
        // Eye tracking direction reads directly from keyboard input dictionary flags
        let playerLookDir = window.keys.right ? 4 : (window.keys.left ? -4 : 0);

        ctx.beginPath();
        ctx.arc(6 + playerLookDir, -player.height / 2 + 5 + playerBob, 6, 0, Math.PI * 2);  // Right socket
        ctx.arc(-6 + playerLookDir, -player.height / 2 + 5 + playerBob, 6, 0, Math.PI * 2); // Left socket
        ctx.fill();
        ctx.stroke();

        // Tiny Focal Pupils
        ctx.fillStyle = '#000000';
        ctx.beginPath();
        ctx.arc(8 + playerLookDir, -player.height / 2 + 5 + playerBob, 2, 0, Math.PI * 2);
        ctx.arc(-4 + playerLookDir, -player.height / 2 + 5 + playerBob, 2, 0, Math.PI * 2);
        ctx.fill();

        // Floating Humanoid Hands
        ctx.fillStyle = player.color;
        ctx.beginPath();
        ctx.arc(-15, 10 + playerBob, 5, 0, Math.PI * 2); // Left hand orb
        ctx.arc(15, 10 + playerBob, 5, 0, Math.PI * 2);  // Right hand orb
        ctx.fill();
        ctx.stroke();

        ctx.restore();
    }
